Manny

A small hackable build system for C and C++.

Download 0.3.0-dev ↓ Read the README →

build.elf

c ::= manny.load("c")

hello ::= c.executable({
	name = "hello",
	sources = {
		"main.c",
		"message.c",
	},
})

entries := {}
entries.build = fun() {
	ret manny.build({
		targets = { hello.task },
	})
}

ret entries

manny

> manny

[1/3 succeeded] compile hello: main.c
[2/3 succeeded] compile hello: message.c
[3/3 succeeded] link hello
