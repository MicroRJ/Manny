build.elf / raw tasks

elf.fs.create_directories("build")

compile := {
	name = "compile main",
	command_line = "clang-cl /c main.c /Fobuild/main.obj",
	inputs = { "main.c" },
	outputs = { "build/main.obj" },
}

link := {
	name = "link hello",
	command_line = "clang-cl build/main.obj /Febuild/hello.exe",
	inputs = { "build/main.obj" },
	outputs = { "build/hello.exe" },
	dependencies = { compile },
}

entries := {}
entries.build = fun() {
	ret manny.build({
		targets = { link },
	})
}

ret entries

That is Manny's entire interface: tasks and dependencies.

But build.elf is a program. You can describe your project as data and turn that data into tasks:

build.elf / generated tasks

sources := { "main.c", "message.c" }

compile_tasks := sources:map(fun(source) {
	object := f"build/${source:slice(0, -2)}.obj"
	ret {
		name = f"compile ${source}",
		command_line = f"clang-cl /c ${source} /Fo${object}",
		inputs = { source },
		outputs = { object },
	}
})

For C and C++, Manny also ships with a Meson-like project descriptor. It handles targets, propagation, generated files, compilation, linking, running, and testing:

build.elf / C project

c ::= manny.load("c")

hello := {
	kind = "executable",
	name = "hello",
	sources = { "main.c", "message.c" },
}

project := {
	name = "hello",
	default_target = hello,
	targets = { hello = hello },
}

tasks := c.project_to_tasks(project)

This layer is written in elf too. Change it, replace it, or skip it.
